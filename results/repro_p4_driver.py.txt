import numpy as np

from lsx import LM
from lsx.core import instruments, ledger, registry, reproduce
from lsx.core.checks import CoreError
from lsx.core.reproduce import DEFERRED, FAILED, LOCAL_TOLERANCE, REFUSED, REPRODUCED, Row

t0 = time.time()


def log(m):
    print(f"[{time.time() - t0:7.1f}s] {m}", flush=True)


led = ledger.Ledger()
rows: list[Row] = []
summary: dict = {}
published: list[str] = []


def publish(claim, note, reproduces, logged):
    """Offer a claim to the real ledger and record what happened."""
    if claim is None:
        return None
    refusal = ledger.would_refuse(claim)
    if refusal:
        return refusal
    led.append(claim, note=note, reproduces=reproduces, logged=logged)
    published.append(claim.id)
    return None


lm = LM.from_pretrained("Qwen/Qwen2.5-1.5B")
log("model loaded")

# ---------------------------------------------------------------- h8 ------------------------
log("h8: three-factor battery")
res8 = reproduce.h8(lm, progress=log)
np.savez(WT + "/results/repro_p4_h8_arrays.npz",
         composed=np.asarray(res8["composed"]), composed_none=np.asarray(res8["composed_none"]),
         **{f"B_{n}_{c}": np.asarray(v) for n, d in res8["B"].items() for c, v in d.items()})
log("checkpointed h8 arrays")
claims8, _ = reproduce.h8_claims(res8)
c8 = claims8[0]
V = res8["n_variants"]
comp = float(np.mean(res8["composed"]))
none8 = float(np.mean(res8["composed_none"]))
summary["h8"] = {"composed": comp, "composed_no_patch": none8, "n_variants": V}
ref8 = publish(c8, "h8 composed, re-run through the core (piece 4)", "h8 composed 2.81/18",
               {"composed": 2.81, "no_patch": 9.50})
rows.append(Row(target="h8 three-factor battery, composed", source="h8 (re-verified h39)",
                logged="2.81/18, no-patch 9.50",
                reproduced=f"{comp:.3f}/{V}, no-patch {none8:.3f}",
                tolerance=f"+-{LOCAL_TOLERANCE} (local, §1A)",
                verdict=REPRODUCED if abs(comp - 2.81) <= LOCAL_TOLERANCE else FAILED,
                detail=f"through `composition` against its declared null of {(V + 1) / 2:.2f}; "
                       f"|delta| {abs(comp - 2.81):.3f}"
                       + (f"; ledger: {ref8}" if ref8 else "; PUBLISHED to the ledger"),
                claim_id=c8.id))

for name, logged_v in (("era", 1.25), ("voice", 1.24), ("tense", 1.03)):
    b = res8["B"][name]
    t = float(np.mean(b["factor"]))
    sel = instruments.build("selector", n=400, d=64, n_candidates=3)
    try:
        sel.claim(treatment=reproduce.Measured(np.asarray(b["factor"]), label=f"h8 {name}"),
                  arms={"random": np.asarray(b["rand"]), "no_patch": np.asarray(b["none"])},
                  floor=reproduce.Floor(stimulus=2.0, estimator=2.0),
                  selection=reproduce.Selection(axis=None, rule="pre-registered layer 14"),
                  provenance=dict(res8["stack"].provenance), grid=res8["grid"], stage="h8")
        v, detail = FAILED, "the core ACCEPTED a battery with no permutation arm"
    except CoreError as e:
        v, detail = REFUSED, f"{type(e).__name__}: {str(e).splitlines()[0][:110]}"
    rows.append(Row(target=f"h8 {name} lens", source="h8", logged=f"{logged_v}/3",
                    reproduced=f"{t:.3f}/3", tolerance=f"+-{LOCAL_TOLERANCE}",
                    verdict=v, detail=detail))
    summary.setdefault("h8_lenses", {})[name] = {"value": t, "no_patch": float(np.mean(b["none"])),
                                                 "random": float(np.mean(b["rand"]))}

# ---------------------------------------------------------------- h14 -----------------------

         **{k: np.asarray(v) for k, v in res14.items()
            if isinstance(v, np.ndarray) or (isinstance(v, list) and k.endswith("_cases") is False)})

c14, s14 = reproduce.h14_claim(res14)
summary["h14"] = s14
ref14 = publish(c14, "h14 negative target, re-run through the core (piece 4)",

rows.append(Row(target="h14 era shift, gain over norm-matched pass-through",

                logged="-0.111 (model 0.889 vs pass-through 1.000)",

                           f"{s14['passthrough']:.3f})",
                tolerance=f"+-{s14['tolerance']:.4f} (paraphrase-noise, measured)",
                verdict=REPRODUCED if abs(s14["gain"] - (-0.111)) <= s14["tolerance"] else FAILED,
                detail=f"zero-shift error {s14['zero_shift_error']:.0g}; random arm "

                logged="1.69/6 (n=48), random 3.25, chance 3.5",
                reproduced=f"{s4['role']:.3f}/6 (n={s4['n']}), random {s4['random']:.3f}, "

                tolerance=f"+-{LOCAL_TOLERANCE} (local, §1A)",
                verdict=REPRODUCED if abs(s4["role"] - 1.69) <= LOCAL_TOLERANCE else FAILED,
                detail=f"|delta| {abs(s4['role'] - 1.69):.3f}; the `permutation` arm h4 never had "
                       "is computed here"
                       + (f"; ledger: {ref4}" if ref4 else "; PUBLISHED to the ledger"),
                claim_id=c4.id))

# ---------------------------------------------------------------- h16 -----------------------
log("h16: from the curve computed earlier")

rows.append(reproduce.h29_from_logged())

               "verdict_counts": counts, "published": published,

log("wrote results/repro_summary_p4.json")
